  if (foundRoot) {
    cpy.fleetRoot = foundRoot;
    cpy.isStandaloneMode = false;
    emitConsoleLine(ste, `>> [FLEET:ROOT] Located multi-cluster workspace at ${foundRoot}`);
  } else {
    cpy.fleetRoot = path.resolve(bal.startDir || process.cwd());
    cpy.isStandaloneMode = true;
    emitConsoleLine(
      ste,
      `>> [FLEET:FALLBACK] Standalone single-repo mode active at ${cpy.fleetRoot}`
    );
  }

  if (bal.slv) {
    bal.slv({
      libBit: {
        idx: 'resolve-fleet-root',

  return cpy;
};

/**
 * Task 1.2: Multi-Cluster Package Manifest Sweep
 * Scans ./packages/* locally alongside sibling cluster package directories
 * for package.json files containing a valid "terminal" declaration block.
 */
export const scanFleetPivots = async (
  cpy: LibraryModel,
  bal: { slv?: (val: any) => void } = {},
  ste?: any
): Promise<LibraryModel> => {
  if (!cpy.fleetRoot) {
    await resolveFleetRoot(cpy, {}, ste);
  }

  const candidateRoots: Array<{ clusterLabel: string; packagesParent: string }> = [
    { clusterLabel: 'local', packagesParent: path.join(process.cwd(), 'packages') },
  ];

  if (!cpy.isStandaloneMode && cpy.fleetRoot) {
    // Direct Front Porch packages (camp-candor-cauldron/packages/*)
    candidateRoots.push({
      clusterLabel: 'front-porch',
      packagesParent: path.join(cpy.fleetRoot, 'camp-candor-cauldron', 'packages'),
    });

    // Numbered thematic clusters (00.governance through 05.delivery)
    const numberedClusters = [
      '00.governance',
      '01.canon',
      '02.simulation',
      '03.foundry',
      '04.assets',
      '05.delivery',
    ];

    for (const cluster of numberedClusters) {
      const clusterPath = path.join(cpy.fleetRoot, cluster);
      if (!fs.existsSync(clusterPath) || isTombstonedDirectory(clusterPath)) continue;

      let repoEntries: fs.Dirent[] = [];
      try {
        repoEntries = fs.readdirSync(clusterPath, { withFileTypes: true });
      } catch {
        continue;
      }

      for (const repoEntry of repoEntries) {
        if (!repoEntry.isDirectory()) continue;
        const repoPath = path.join(clusterPath, repoEntry.name);
        if (isTombstonedDirectory(repoPath)) continue;
        candidateRoots.push({
          clusterLabel: `${cluster}/${repoEntry.name}`,
          packagesParent: path.join(repoPath, 'packages'),
        });
      }
    }
  }

  const discoveredMap = new Map<string, DiscoveredPivotBit>();

  for (const candidate of candidateRoots) {
    if (!fs.existsSync(candidate.packagesParent)) continue;

    let pkgEntries: fs.Dirent[] = [];
    try {
      pkgEntries = fs.readdirSync(candidate.packagesParent, { withFileTypes: true });
    } catch {
      continue;
    }

    for (const pkgEntry of pkgEntries) {
      if (!pkgEntry.isDirectory()) continue;
      const pkgDir = path.join(candidate.packagesParent, pkgEntry.name);
      if (isTombstonedDirectory(pkgDir)) continue;

      const pkgJsonPath = path.join(pkgDir, 'package.json');
      if (!fs.existsSync(pkgJsonPath)) continue;

      try {
        const raw = fs.readFileSync(pkgJsonPath, 'utf8');
        const pkgJson = JSON.parse(raw);
        const terminalConfig = parseTerminalConfig(pkgJson);
        if (!terminalConfig) continue;

        // Avoid re-registering 995.library itself as an external pivot
        if (terminalConfig.globalKey === 'LIBRARY') continue;

        // Resolve compiled hunt.js location (dist/hunt.js or hunt.js)
        const distHunt = path.join(pkgDir, 'dist', 'hunt.js');
        const rootHunt = path.join(pkgDir, 'hunt.js');
        const huntEntryPath = fs.existsSync(distHunt) ? distHunt : rootHunt;

        if (!discoveredMap.has(terminalConfig.globalKey)) {
          discoveredMap.set(terminalConfig.globalKey, {
            packageName: pkgJson.name || pkgEntry.name,
            packageDir: pkgDir,
            clusterLabel: candidate.clusterLabel,
            huntEntryPath,
            terminal: terminalConfig,
          });
        }
      } catch (err: any) {
        emitConsoleLine(
          ste,
          `:: [WARN] Malformed package.json skipped at ${pkgJsonPath}: ${err?.message || 'parse error'}`
        );
      }
    }
  }

  cpy.discoveredPivots = Array.from(discoveredMap.values());
  emitConsoleLine(
    ste,
    `>> [FLEET:SCAN] Discovered ${cpy.discoveredPivots.length} external terminal pivot(s)`
  );

  if (bal.slv) {
    bal.slv({
      libBit: {
        idx: 'scan-fleet-pivots',
        count: cpy.discoveredPivots.length,
        pivots: cpy.discoveredPivots,
      },
    });
  }
  return cpy;
};

/**
 * Task 1.3: Fail-Closed Pivot Module Hydration
 * Dynamically imports each discovered package's compiled hunt.js inside an isolated
 * try/catch boundary and binds the resolved state machine onto globalThis[globalKey]
 * and globalThis.PIVOTS[packageName].
 */
export const hydratePivotModules = async (
  cpy: LibraryModel,
  bal: { registerMenus?: boolean; slv?: (val: any) => void } = {},
  ste?: any
): Promise<LibraryModel> => {
  const g = globalThis as Record<string, any>;
  if (!g.PIVOTS || typeof g.PIVOTS !== 'object') {
    g.PIVOTS = {};
  }

  const receipts: PivotHydrationReceipt[] = [];

  for (const pivot of cpy.discoveredPivots) {
    if (!fs.existsSync(pivot.huntEntryPath)) {
      const receipt: PivotHydrationReceipt = {
        globalKey: pivot.terminal.globalKey,
        menuTitle: pivot.terminal.menuTitle,
        packageName: pivot.packageName,
        status: 'SKIPPED_NO_BUILD',
        reason: `Missing compiled hunt.js at ${pivot.huntEntryPath}`,
      };
      receipts.push(receipt);
      emitConsoleLine(
        ste,
        `:: [SKIP] ${pivot.terminal.menuTitle} (${pivot.packageName}) -> hunt.js not compiled`
      );
      continue;
    }

    try {
      const fileUrl = pathToFileURL(pivot.huntEntryPath).href;
      const importedMod = await import(fileUrl);
      const pivotInstance =
        importedMod?.default ||
        importedMod?.[pivot.terminal.globalKey] ||
        importedMod?.sim ||
        importedMod;

      if (!pivotInstance || typeof pivotInstance.hunt !== 'function') {
        throw new Error('Exported module does not expose a valid hunt() state machine interface');
      }

      g[pivot.terminal.globalKey] = pivotInstance;
      g.PIVOTS[pivot.packageName] = pivotInstance;

      const receipt: PivotHydrationReceipt = {
        globalKey: pivot.terminal.globalKey,
        menuTitle: pivot.terminal.menuTitle,
        packageName: pivot.packageName,
        status: 'HYDRATED',
      };
      receipts.push(receipt);
      emitConsoleLine(
        ste,
        `:: [OK] Hydrated ${pivot.terminal.menuTitle} -> globalThis.${pivot.terminal.globalKey} (${pivot.clusterLabel})`
      );

      if (bal.registerMenus !== false && ste && typeof ste.hunt === 'function') {
        await ste.hunt('[Menu action] Route Menu', {
          menuTitle: pivot.terminal.menuTitle,
          menuDesc: pivot.terminal.menuDesc,
          globalKey: pivot.terminal.globalKey,
        });
      }
    } catch (err: any) {
      const reason = err?.message || String(err);
      const receipt: PivotHydrationReceipt = {
        globalKey: pivot.terminal.globalKey,
        menuTitle: pivot.terminal.menuTitle,
        packageName: pivot.packageName,
        status: 'FAILED',
        reason,
      };
      receipts.push(receipt);
      emitConsoleLine(
        ste,
        `:: [FAIL] Isolated hydration failure for ${pivot.terminal.menuTitle}: ${reason}`
      );
    }
  }

  cpy.hydrationReceipts = receipts;
  if (bal.slv) {
    bal.slv({
      libBit: {
        idx: 'hydrate-pivot-modules',
        receipts,
      },
    });
  }
  return cpy;
};

/**
 * Combined Discovery Pipeline (Resolve -> Scan -> Hydrate)
 */
export const discoverFleetPivots = async (
  cpy: LibraryModel,
  bal: { startDir?: string; registerMenus?: boolean; slv?: (val: any) => void } = {},
  ste?: any
): Promise<LibraryModel> => {
  emitConsoleLine(ste, '>> [DISCOVERY] Initiating Local Multi-Cluster Pivot Discovery...');
  await resolveFleetRoot(cpy, { startDir: bal.startDir }, ste);
  await scanFleetPivots(cpy, {}, ste);
  await hydratePivotModules(cpy, { registerMenus: bal.registerMenus }, ste);

  const hydratedCount = cpy.hydrationReceipts.filter((r) => r.status === 'HYDRATED').length;
  emitConsoleLine(
    ste,
    `>> [DISCOVERY:COMPLETE] ${hydratedCount}/${cpy.discoveredPivots.length} pivot session(s) online.`
  );

  const summary: FleetDiscoverySummary = {
    fleetRoot: cpy.fleetRoot,
    isStandalone: cpy.isStandaloneMode,
    discovered: cpy.discoveredPivots,
    receipts: cpy.hydrationReceipts,
  };

  if (bal.slv) {
    bal.slv({ libBit: { idx: 'discover-fleet-pivots', summary } });
  }
  return cpy;
};
```

import type { ConsoleModel } from '../console.model.js';

export interface ConsoleBit {
  idx?: string;
  src?: string;
  dat?: {
    preserveHistory?: boolean;
    sessionNamespace?: string;
    clearExplicit?: boolean;
  };
  slv?: (val: { cnsBit: Partial<ConsoleBit> }) => void;
}

function sanitizeAscii(input: string): string {
  let clean = '';
  for (let i = 0; i < input.length; i++) {
    const code = input.charCodeAt(i);
    if (code >= 0x20 && code <= 0x7e) {
      clean += input[i];
    }
  }
  return clean;
}

/**
 * Task 4.3: Namespaced Console Telemetry Formatting & Scrollback Preservation
 * Preserves cns00 history across session switches unless clearExplicit === true,
 * and formats session engagement banners cleanly in 7-bit ASCII.
 */
export const updateConsole = async (
  cpy: ConsoleModel,
  bal: ConsoleBit,
  ste?: any
): Promise<ConsoleModel> => {
  const paneIdx = bal?.idx || 'cns00';

  if (bal?.dat?.clearExplicit === true) {
    cpy.scrollbackLines = [];
  }

  if (typeof bal?.dat?.sessionNamespace === 'string' && bal.dat.sessionNamespace.trim().length > 0) {
    cpy.activeNamespace = sanitizeAscii(bal.dat.sessionNamespace.trim());
  }

  const rawSrc = typeof bal?.src === 'string' ? bal.src : '';
  const sanitizedLine = sanitizeAscii(rawSrc);

  if (sanitizedLine.length > 0) {
    cpy.scrollbackLines.push(sanitizedLine);
    if (cpy.scrollbackLines.length > cpy.maxScrollbackLines) {
      cpy.scrollbackLines.splice(0, cpy.scrollbackLines.length - cpy.maxScrollbackLines);
    }
  }

  const g = globalThis as Record<string, any>;
  if (ste && g.LIBRARY && typeof g.LIBRARY.hunt === 'function' && sanitizedLine.length > 0) {
    await g.LIBRARY.hunt('[Text action] Update Text', {
      idx: paneIdx,
      src: sanitizedLine,
    });
  }

  if (bal?.slv) {
    bal.slv({
      cnsBit: {
        idx: paneIdx,
        src: sanitizedLine,
      },
    });
  }

  return cpy;
};

/**
 * Task 5.1: Boot Sequence Orchestration in Library Unit
 * Chains 80.terminal.unit initialization, scanFleetPivots, hydratePivotModules,
 * 98.menu.unit ROUTE_MENU registration, and Master Switchboard launch in deterministic order.
 */
export const bootSwitchboard = async (
  cpy: LibraryModel,
  bal: BootSwitchboardPayload = {},
  ste?: any
): Promise<LibraryModel> => {
  const isHeadless = bal?.headless === true || !ste || typeof ste.hunt !== 'function';

  // Step 1: Initialize 80.terminal.unit & seed shared Master Switchboard HUD context
  if (!isHeadless) {
    await ste.hunt('[Terminal action] Init Terminal', {});
    await ste.hunt('[Terminal action] Set Session Context', {
      activePivotTitle: 'MASTER SWITCHBOARD',
      hostRigRole: process.env.HOST_RIG_ROLE || 'RIG 1 (ORCHESTRATOR)',
    });
  }

  emitBootConsole(ste, '>> [BOOT:SWITCHBOARD] Initiating unified local workstation harness...');

  // Step 2: Resolve fleet root and sweep multi-cluster packages
  await resolveFleetRoot(cpy, { startDir: bal?.startDir }, ste);
  await scanFleetPivots(cpy, {}, ste);

  // Step 3: Hydrate pivot modules (with registerMenus: false so Step 4 controls exact route order)
  await hydratePivotModules(cpy, { registerMenus: false }, ste);

  // Step 4: Deterministically register every HYDRATED pivot with 98.menu.unit via ROUTE_MENU
  const hydratedPivots = cpy.discoveredPivots.filter((pivot) =>
    cpy.hydrationReceipts.some(
      (r) => r.globalKey === pivot.terminal.globalKey && r.status === 'HYDRATED'
    )
  );

  const mountedTitles: string[] = ['LIBRARY MENU'];

  for (const pivot of hydratedPivots) {
    if (ste && typeof ste.hunt === 'function') {
      await ste.hunt('[Menu action] Route Menu', {
        menuTitle: pivot.terminal.menuTitle,
        menuDesc: pivot.terminal.menuDesc,
        globalKey: pivot.terminal.globalKey,
      });
    }
    mountedTitles.push(pivot.terminal.menuTitle);
  }

  const receipt: BootSwitchboardReceipt = {
    fleetRoot: cpy.fleetRoot,
    isStandalone: cpy.isStandaloneMode,
    mountedCount: hydratedPivots.length,
    totalDiscovered: cpy.discoveredPivots.length,
    mountedTitles,
  };

  cpy.switchboardBooted = true;
  cpy.lastBootReceipt = receipt;

  emitBootConsole(
    ste,
    `>> [BOOT:READY] Mounted ${mountedTitles.length} switchboard route(s): ${mountedTitles.join(', ')}`
  );

  // Step 5: Open the Master Root Switchboard interactive loop when running with live TUI
  if (!isHeadless) {
    await ste.hunt('[Menu action] Init Menu', { slv: bal?.slv });
  } else if (bal?.slv) {
    bal.slv({
      libBit: {
        idx: 'boot-switchboard-headless',
        receipt,
      },
    });
  }

  return cpy;
};
```

export function buildSessionRouteAuditTable(
  menuRoutes: Map<string, any>
): { rows: SessionRouteAuditRow[]; asciiLines: string[] } {
  const g = globalThis as Record<string, any>;
  const rows: SessionRouteAuditRow[] = [
    {
      menuTitle: 'LIBRARY MENU',
      globalKey: 'LIBRARY',
      routeRegistered: true,
      globalBound: typeof g.LIBRARY?.hunt === 'function',
      status: typeof g.LIBRARY?.hunt === 'function' ? 'READY' : 'DEGRADED',
    },
  ];

  // Cross-check registered external routes against globalThis bindings
  for (const [menuTitle] of menuRoutes.entries()) {
    const inferredKey = menuTitle
      .replace(/\s+MENU$/i, '')
      .replace(/[^A-Z0-9]+/gi, '_')
      .toUpperCase();
    const matchedKey =
      Object.keys(g).find(
        (k) =>
          (k === inferredKey || inferredKey.endsWith(k)) &&
          g[k] &&
          typeof g[k].hunt === 'function'
      ) || inferredKey;

    const globalBound = Boolean(g[matchedKey] && typeof g[matchedKey].hunt === 'function');
    rows.push({
      menuTitle,
      globalKey: matchedKey,
      routeRegistered: true,
      globalBound,
      status: globalBound ? 'READY' : 'UNBOUND',
    });
  }

  const asciiLines: string[] = [
    '>> +--------------------------+--------------+---------+---------+',
    '>> | ROUTE TITLE              | GLOBAL KEY   | ROUTE   | STATUS  |',
    '>> +--------------------------+--------------+---------+---------+',
  ];

  for (const row of rows) {
    const colTitle = row.menuTitle.padEnd(24, ' ').slice(0, 24);
    const colKey = row.globalKey.padEnd(12, ' ').slice(0, 12);
    const colRoute = (row.routeRegistered ? 'MOUNTED' : 'MISSING').padEnd(7, ' ');
    const colStatus = row.status.padEnd(7, ' ');
    asciiLines.push(`>> | ${colTitle} | ${colKey} | ${colRoute} | ${colStatus} |`);
  }

  asciiLines.push('>> +--------------------------+--------------+---------+---------+');
  return { rows, asciiLines };
}

function collectRegisteredRoutes(): SwitchboardRouteManifestItem[] {
  const g = globalThis as Record<string, any>;
  const pivots = g.PIVOTS && typeof g.PIVOTS === 'object' ? g.PIVOTS : {};
  const seenKeys = new Set<string>();
  const routes: SwitchboardRouteManifestItem[] = [];

  for (const [entryKey, instance] of Object.entries(pivots)) {
    if (!instance || typeof instance !== 'object') continue;
    const mod = instance as Record<string, any>;
    const globalKey =
      typeof mod.globalKey === 'string' && mod.globalKey.trim().length > 0
        ? mod.globalKey.trim().toUpperCase()
        : entryKey.trim().toUpperCase();

    if (seenKeys.has(globalKey)) continue;
    seenKeys.add(globalKey);

    routes.push({
      globalKey,
      menuTitle:
        typeof mod.menuTitle === 'string' && mod.menuTitle.trim().length > 0
          ? mod.menuTitle.trim()
          : `${globalKey} MENU`,
      menuDesc:
        typeof mod.menuDesc === 'string' && mod.menuDesc.trim().length > 0
          ? mod.menuDesc.trim()

      activeSeriesSlug:
        typeof mod.activeSeriesSlug === 'string' && mod.activeSeriesSlug.trim().length > 0
          ? mod.activeSeriesSlug.trim()
          : 'UNBOUND',
    });
  }

  for (const fallback of DEFAULT_FALLBACK_ROUTES) {
    if (!seenKeys.has(fallback.globalKey)) {
      seenKeys.add(fallback.globalKey);
      routes.unshift(fallback);
    }
  }

  return routes;
