```typescript
      rootSlv({ mnuBit: { idx: 'exit-switchboard' } });
    }
    await ste.hunt(ActTrm.CLOSE_TERMINAL, {});
    return cpy;
  }

  if (src === DEFAULT_LIBRARY_MENU_TITLE || cpy.menuRoutes.has(src)) {
    // Task 3.1 & 3.2: Clear grid, yield in-process, await child bal.slv resolution
    await yieldToPivotRoute(cpy, src, ste);

    // Restore Master Switchboard state immediately upon child submenu ROOT MENU return
    cpy.activeMenuMode = 'ROOT';
    cpy.activeSessionTitle = 'MASTER SWITCHBOARD';

    await ste.hunt(ActCns.UPDATE_CONSOLE, {
      idx: 'cns00',
      src: '>> [RETURN] Restored Master Root Switchboard.',
    });
  }

  setTimeout(async () => {
    await ste.hunt(ActMnu.UPDATE_MENU, {});
  }, 100);

  return cpy;
};
```

```typescript
// Excerpt for packages/995.library/98.menu.unit/buz/menu.library.ts
import * as ActMnu from '../menu.action.js';
import * as ActTrm from '../../80.terminal.unit/terminal.action.js';
import * as ActGrd from '../../81.grid.unit/grid.action.js';
import * as ActCns from '../../83.console.unit/console.action.js';
import * as ActChc from '../../85.choice.unit/choice.action.js';
import * as Align from '../../val/align.js';
import * as Color from '../../val/console-color.js';
import type { MenuModel } from '../menu.model.js';
import type MenuBit from '../fce/menu.bit.js';

export const libraryMenu = async (
  cpy: MenuModel,
  bal: MenuBit = { idx: 'library-menu' },
  ste?: any
): Promise<MenuModel> => {
  // Task 3.2: Headless safety guard
  if (!ste) {
    if (bal?.slv) {
      bal.slv({ mnuBit: { idx: 'library-menu-headless' } });
    }
    return cpy;
  }

  const lst = [
    'DISCOVER FLEET PIVOTS',
    'LIST SWITCHBOARD ROUTES',
    'JUMP TO MOUNTED PIVOT',
    'ROOT MENU',
  ];

  const gridBit = await ste.hunt(ActGrd.UPDATE_GRID, {
    x: 0,
    y: 4,
    xSpan: 4,
    ySpan: 8,
  });

  const choiceBit = await ste.hunt(ActChc.OPEN_CHOICE, {
    dat: {
      clr0: Color.BLACK,
      clr1: Color.GREEN,
      cb: (choice: string) => {
        ste.hunt(ActCns.UPDATE_CONSOLE, {
          idx: 'cns00',
          src: `:: [LIBRARY] ${choice}`,
        });
      },
    },
    src: Align.VERTICAL,
    lst,
    net: gridBit?.grdBit?.dat,
  });

  const src: string = choiceBit?.chcBit?.src || 'ROOT MENU';

  switch (src) {
    case 'DISCOVER FLEET PIVOTS': {
      await ste.hunt('[Library action] Discover Fleet Pivots', { registerMenus: true });
      break;
    }

    case 'LIST SWITCHBOARD ROUTES': {
      await ste.hunt(ActCns.UPDATE_CONSOLE, {
        idx: 'cns00',
        src: '>> [SWITCHBOARD] Registered Master Root Routes:',
      });
      for (const title of ['LIBRARY MENU', ...Array.from(cpy.menuRoutes.keys())]) {
        const desc = cpy.menuRouteDescriptions.get(title) || 'No description';
        await ste.hunt(ActCns.UPDATE_CONSOLE, {
          idx: 'cns00',
          src: `:: [${title}] -> ${desc.replace(/\s+/g, ' ')}`,
        });
      }
      break;
    }

    case 'JUMP TO MOUNTED PIVOT': {
      const externalTargets = Array.from(cpy.menuRoutes.keys());
      if (externalTargets.length === 0) {
        await ste.hunt(ActCns.UPDATE_CONSOLE, {
          idx: 'cns00',
          src: ':: [WARN] No external pivots mounted to jump into.',
        });
        break;
      }
      await ste.hunt(ActTrm.CLEAR_TERMINAL, {});
      const jumpChoice = await ste.hunt(ActChc.OPEN_CHOICE, {
        dat: { clr0: Color.BLACK, clr1: Color.CYAN },
        src: Align.VERTICAL,
        lst: [...externalTargets, 'CANCEL'],
        net: gridBit?.grdBit?.dat,
      });
      const target = jumpChoice?.chcBit?.src || 'CANCEL';
      if (target !== 'CANCEL') {
        await ste.hunt(ActMnu.JUMP_PIVOT, { targetMenuTitle: target });
        if (bal?.slv) {
          bal.slv({ mnuBit: { idx: 'root-menu-via-jump', src: target } });
        }
        return cpy;
      }
      break;
    }

    case 'ROOT MENU':
    default: {
      // Task 3.2: Resolve child promise cleanly without closing blessed.screen
      await ste.hunt(ActTrm.CLEAR_TERMINAL, {});
      if (bal?.slv) {
        bal.slv({ mnuBit: { idx: 'root-menu' } });
      }
      return cpy;
    }
  }

  return libraryMenu(cpy, bal, ste);
};
```
