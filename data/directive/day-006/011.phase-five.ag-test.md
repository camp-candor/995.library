```bash
npx tsx -e "
import fs from 'node:fs';
import os from 'node:os';
import path from 'node:path';
import { LibraryModel } from './packages/995.library/00.library.unit/library.model.js';
import { bootSwitchboard } from './packages/995.library/00.library.unit/buz/boot-switchboard.buzz.js';
import { buildSessionRouteAuditTable } from './packages/995.library/98.menu.unit/buz/menu.library.js';

async function runProbe() {
  const tmpRoot = fs.mkdtempSync(path.join(os.tmpdir(), 'ag-phase5-'));
  try {
    const cauldronPkg = path.join(tmpRoot, 'camp-candor-cauldron', 'packages', '000.cauldron');
    const lorePkg = path.join(tmpRoot, '01.canon', '000.goblin-lore', 'packages', '299.lore');
    fs.mkdirSync(cauldronPkg, { recursive: true });
    fs.mkdirSync(lorePkg, { recursive: true });

    fs.writeFileSync(path.join(cauldronPkg, 'package.json'), JSON.stringify({
      name: '@camp_candor/000.cauldron',
      terminal: { globalKey: 'CAULDRON', menuTitle: 'CAULDRON MENU', menuDesc: 'Cauldron test pivot' }
    }));
    fs.writeFileSync(path.join(cauldronPkg, 'hunt.js'), 'export default { hunt: async () => ({ ok: true }) };');

    fs.writeFileSync(path.join(lorePkg, 'package.json'), JSON.stringify({
      name: '@camp_candor/299.lore',
      terminal: { globalKey: 'LORE', menuTitle: 'GOBLIN LORE MENU', menuDesc: 'Lore test pivot' }
    }));
    fs.writeFileSync(path.join(lorePkg, 'hunt.js'), 'export default { hunt: async () => ({ ok: true }) };');

    const callSequence: string[] = [];
    const mountedRoutes = new Map<string, any>();

    const mockSte = {
      hunt: async (type: string, bal: any) => {
        callSequence.push(type);
        if (type === '[Menu action] Route Menu' && bal?.menuTitle) {
          mountedRoutes.set(bal.menuTitle, () => Promise.resolve());
        }
        return {};
      }
    };

    (globalThis as any).LIBRARY = { hunt: async () => ({ ok: true }) };

    const model = new LibraryModel();
    await bootSwitchboard(model, {
      startDir: path.join(tmpRoot, 'camp-candor-cauldron'),
      headless: true,
    }, mockSte);

    const routeCalls = callSequence.filter((c) => c === '[Menu action] Route Menu');
    if (
      !model.switchboardBooted ||
      !model.lastBootReceipt ||
      model.lastBootReceipt.mountedCount !== 2 ||
      routeCalls.length !== 2 ||
      !mountedRoutes.has('CAULDRON MENU') ||
      !mountedRoutes.has('GOBLIN LORE MENU')
    ) {
      console.error(':: [FAIL] Deterministic boot sequence or route registration mismatch:', {
        switchboardBooted: model.switchboardBooted,
        lastBootReceipt: model.lastBootReceipt,
        callSequence,
      });
      process.exit(1);
    }

    const { rows, asciiLines } = buildSessionRouteAuditTable(mountedRoutes);
    const allReady = rows.every((r) => r.status === 'READY');
    const hasAsciiFrame = asciiLines.some((l) => l.includes('+--------------------------+'));

    if (!allReady || !hasAsciiFrame) {
      console.error(':: [FAIL] Session route readiness table audit mismatch:', { rows, asciiLines });
      process.exit(1);
    }

    console.log('>> [OK] Deterministic boot sequence and 7-bit ASCII readiness table verified.');
    process.exit(0);
  } finally {
    fs.rmSync(tmpRoot, { recursive: true, force: true });
  }
}
runProbe();
"
```
