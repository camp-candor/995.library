### Level 3: Synthetic In-Memory Fleet Discovery & Fail-Closed Probe

Execute a live synthetic runtime probe against `discoverFleetPivots` using temporary scratch directories to confirm end-to-end discovery, isolation, and cleanup.

* [ ] **3.1 Synthetic Multi-Cluster Discovery & Isolation Probe**
* **Command:**
```bash
npx tsx -e "
import fs from 'node:fs';
import os from 'node:os';
import path from 'node:path';
import { LibraryModel } from './packages/995.library/00.library.unit/library.model.js';
import { discoverFleetPivots } from './packages/995.library/00.library.unit/buz/pivot-discovery.buzz.js';

async function runProbe() {
  const tmpRoot = fs.mkdtempSync(path.join(os.tmpdir(), 'ag-phase1-'));
  try {
    // Create multi-cluster markers and valid + unbuilt sibling pivots
    const cauldronPkg = path.join(tmpRoot, 'camp-candor-cauldron', 'packages', '000.cauldron');
    const lorePkg = path.join(tmpRoot, '01.canon', '000.goblin-lore', 'packages', '299.lore');
    const brokenPkg = path.join(tmpRoot, '02.simulation', '000.series-sower-engine', 'packages', '300.sower');

    fs.mkdirSync(cauldronPkg, { recursive: true });
    fs.mkdirSync(lorePkg, { recursive: true });
    fs.mkdirSync(brokenPkg, { recursive: true });

    fs.writeFileSync(path.join(cauldronPkg, 'package.json'), JSON.stringify({
      name: '@camp_candor/000.cauldron',
      terminal: { globalKey: 'CAULDRON', menuTitle: 'CAULDRON MENU', menuDesc: 'Cauldron test pivot' }
    }));
    fs.writeFileSync(path.join(cauldronPkg, 'hunt.js'), 'export default { hunt: async () => ({ ok: true }) };');

    fs.writeFileSync(path.join(lorePkg, 'package.json'), JSON.stringify({
      name: '@camp_candor/299.lore',
      terminal: { globalKey: 'LORE', menuTitle: 'GOBLIN LORE MENU', menuDesc: 'Lore test pivot' }
    }));
    fs.writeFileSync(path.join(lorePkg, 'hunt.js'), 'throw new Error(\"Simulated broken pivot\");');

    fs.writeFileSync(path.join(brokenPkg, 'package.json'), JSON.stringify({
      name: '@camp_candor/300.sower',
      terminal: { globalKey: 'SOWER', menuTitle: 'SERIES SOWER MENU', menuDesc: 'Unbuilt test pivot' }
    }));

    const model = new LibraryModel();
    let captured = null;
    await discoverFleetPivots(model, {
      startDir: path.join(tmpRoot, 'camp-candor-cauldron'),
      registerMenus: false,
      slv: (res) => { captured = res; }
    });

    const receipts = model.hydrationReceipts;
    const hydrated = receipts.find(r => r.globalKey === 'CAULDRON' && r.status === 'HYDRATED');
    const failed = receipts.find(r => r.globalKey === 'LORE' && r.status === 'FAILED');
    const skipped = receipts.find(r => r.globalKey === 'SOWER' && r.status === 'SKIPPED_NO_BUILD');

    if (!hydrated || !failed || !skipped || typeof globalThis.CAULDRON?.hunt !== 'function') {
      console.error(':: [FAIL] Probe assertion mismatch:', { receipts, captured });
      process.exit(1);
    }
    console.log('>> [OK] Synthetic multi-cluster discovery & fail-closed isolation verified.');
  } finally {
    fs.rmSync(tmpRoot, { recursive: true, force: true });
  }
}
runProbe();
"
```
