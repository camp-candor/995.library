### I. Architectural Context & Invariants

1. **The 12×12 Percentage Coordinate Matrix:**
* In `apps/995.library`, terminal windows are mapped into a rigid 12-column by 12-row percentage grid via `81.grid.unit` (`UPDATE_GRID`).


* The workspace is vertically bisected into two dedicated operational axes:


* **Control Axis (Left, Columns 0–4, Rows 0–12):** 33.33% width dedicated to navigation menus (`98.menu.unit`), choices (`85.choice.unit`), and input modals (`84.input.unit`).


* **Telemetry Axis (Right, Columns 4–12, Rows 0–12):** 66.67% width dedicated to system observability.


* **Status HUD (Top Right, Columns 4–12, Rows 0–3):** 25.00% height for active target/series and Merkle hashes via `80.terminal.unit`.


* **Console Log (Bottom Right, Columns 4–12, Rows 3–12):** 75.00% height for the `cns00` live scrollable log viewport via `83.console.unit`.






2. **TTY Raw Mode & Alternate Screen Buffer:**
* When `80.terminal.unit` initializes, the terminal process takes ownership of standard input via `process.stdin.setRawMode(true)` if running in a TTY environment.


* The screen runtime writes the alternate buffer escape code (`\x1b[?1049h`), redrawing the 12×12 grid directly into memory and isolating standard output to prevent curses buffer tearing.




3. **Repository Laws (`AGENTS.md`):**
* Immutability on `apps/995.library/995.library/**` is **permanently lifted** in this repository.


* All console outputs, statuses, and layout labels must observe **7-bit pure ASCII** (`>>`, `[OK]`, `[FAIL]`, `::`, `---`).


* Headless test execution must degrade gracefully without throwing when Blessed screen instances are absent.

import type { GridModel } from '../grid.model'
import type GridBit from '../fce/grid.bit'
import type State from '../../99.core/state'
import type { TerminalModel } from '../../80.terminal.unit/terminal.model'
import type NetBit from '../fce/net.bit'

export const initGrid = (cpy: GridModel, bal: GridBit, _ste: State) => {
    if (bal?.slv != null) bal.slv({ grdBit: { idx: 'init-grid' } })
    return cpy
}

export const updateGrid = (cpy: GridModel, bal: GridBit, ste: State) => {
    const termMod: TerminalModel | undefined = ste?.value?.terminal

    const margin = 0
    const cols = termMod?.cols && termMod.cols > 0 ? termMod.cols : 12
    const rows = termMod?.rows && termMod.rows > 0 ? termMod.rows : 12

    const colNow = typeof bal?.x === 'number' ? bal.x : 0
    const rowNow = typeof bal?.y === 'number' ? bal.y : 0

    const colSpan = typeof bal?.xSpan === 'number' ? bal.xSpan : 12
    const rowSpan = typeof bal?.ySpan === 'number' ? bal.ySpan : 12

    const spacing = 0

    const cellWidth = (100 - margin * 2) / cols
    const cellHeight = (100 - margin * 2) / rows

    const top = `${rowNow * cellHeight + margin}%`
    const left = `${colNow * cellWidth + margin}%`
    const width = `${cellWidth * colSpan - spacing}%`
    const height = `${cellHeight * rowSpan - spacing}%`

    const bit: NetBit = { left, top, width, height }

    if (bal?.slv != null) bal.slv({ grdBit: { idx: 'update-grid', dat: bit } })

    return cpy
}

export const initTerminal = async (
    cpy: TerminalModel,
    bal: TerminalBit,
    ste: State,
) => {
    if (firstLoad) return cpy
    firstLoad = true

    if (bal?.dat != null) {
        bit = await ste.hunt(ActBus.INIT_BUS, {
            idx: cpy.idx,
            lst: [ActTrm, ActChc, ActPut, ActGrd, ActCns],
            dat: bal.dat,
            src: bal.src,
        })
    }

    bit = await ste.hunt(ActTrm.OPEN_TERMINAL, {})

    if (bal?.val === 1) patch(ste, ActMnu.INIT_MENU, bal)
    if (bal?.slv != null) bal.slv({ trmBit: { idx: 'init-terminal' } })

    return cpy
}

    cpy: TerminalModel,
    bal: TerminalBit,
    _ste: State,
) => {
    const lstMsg: any[] = []
    if (bal?.slv != null) {
        bal.slv({ trmBit: { idx: 'update-terminal', lst: lstMsg } })
    }
    return cpy
}

export const openTerminal = async (
    cpy: TerminalModel,
    bal: TerminalBit,
    _ste: State,
) => {
    const blessed = (cpy.blessed = require('blessed'))
    cpy.contrib = require('blessed-contrib')

    if (process.stdin.isTTY && typeof process.stdin.setRawMode === 'function') {
        try {
            process.stdin.setRawMode(true)
        } catch {
            // Non-blocking degradation in test environments
        }
    }

    const screen = (cpy.screen = blessed.screen({
        smartCSR: true,
        title: '995.LIBRARY // TERMINAL FLIGHT DECK',
        cursor: { artificial: true, shape: 'line', blink: true, color: 'green' },
        fullUnicode: false,
        warnings: false,
    }))

    screen.render()

    if (bal?.slv != null) bal.slv({ trmBit: { idx: 'open-terminal' } })
    return cpy
}

export const closeTerminal = (
    cpy: TerminalModel,
    bal: TerminalBit,
    _ste: State,
) => {
    if (cpy.screen != null) {
        try {
            if (typeof cpy.screen.leave === 'function') {
                cpy.screen.leave()
            }
            cpy.screen.destroy()
        } catch {
            // Ignore teardown errors
        }
    }

    cpy.blessed = null
    cpy.contrib = null
    cpy.screen = null

    if (bal?.slv != null) bal.slv({ trmBit: { idx: 'close-terminal' } })
    return cpy
}

    bal: TerminalBit,
    _ste: State,

    if (cpy.blessed) {

            cpy.blessed.program().clear()
            if (cpy.screen && typeof cpy.screen.realloc === 'function') {

    return cpy
}

export const layoutTerminal = (
    cpy: TerminalModel,

    _ste: State,

            break

        case Grid.MID_FULL_IDX:
            fillBit = Grid.MID_FULL_BIT
            break

        case Grid.TOP_FULL_IDX:
            fillBit = Grid.TOP_FULL_BIT
            break

        case Grid.CONTROL_AXIS_IDX:
            fillBit = Grid.CONTROL_AXIS_BIT
            break

        case Grid.TELEMETRY_AXIS_IDX:
            fillBit = Grid.TELEMETRY_AXIS_BIT
            break

        case Grid.STATUS_HUD_IDX:
            fillBit = Grid.STATUS_HUD_BIT
            break

        case Grid.CONSOLE_LOG_IDX:
            fillBit = Grid.CONSOLE_LOG_BIT
            break
    }

    if (bal?.slv != null) {
        bal.slv({ trmBit: { idx: 'layout-terminal', dat: fillBit } })
