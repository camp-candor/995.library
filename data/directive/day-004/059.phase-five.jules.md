#### The Five Operational Remediation Vectors:

1. **1. RETRY FAILED TARGET:** Extracts failed targets from `sagaJournal.targetsFailed`, prompts the operator to unlock processes, and re-dispatches `PROGRESS_LIBRARY_SAGA` targeting only the failed nodes.


2. **2. SKIP TARGET & RESUME SAGA:** Flags the failed target as `DESYNCHRONIZED` in memory, purges orphaned `.tmp_staging_*` buffers, and advances to remaining targets.


3. **3. COMPENSATE & ROLLBACK COMPLETED:** Iterates `sagaJournal.targetsCompleted` in reverse chronological order, restoring each completed repository from its preserved `.tmp_backup_*` directory via atomic rename, transitioning the fleet back to a clean pre-saga baseline.


4. **4. ABORT SAGA & PRESERVE PARTIAL STATE:** Freezes the saga journal with status `'TORN'`, leaves successfully promoted nodes live, and unlinks temporary staging buffers on failed nodes.


5. **5. DUMP FORENSIC SAGA DOSSIER TO cns00:** Serializes the full `SagaJournal` transaction log, target paths, backup mappings, and error stacks, streaming them asynchronously to `cns00` without blocking the curses event loop.

import clone from 'clone-deep'
import * as Act from './menu.action'

import * as Buzz from './menu.buzzer'
import type State from '../99.core/state'

export function reducer(
    model: MenuModel = new MenuModel(),
    act: Act.Actions,
    state?: State,
) {
    switch (act.type) {
        case Act.UPDATE_MENU:
            return Buzz.updateMenu(clone(model, true), act.bale, state)

        case Act.INIT_MENU:
            return Buzz.initMenu(clone(model, true), act.bale, state)

        case Act.OPEN_MENU:
            return Buzz.openMenu(clone(model, true), act.bale, state)

        case Act.CLOSE_MENU:
            return Buzz.closeMenu(clone(model, true), act.bale, state)

        case Act.PRINT_MENU:
            return Buzz.printMenu(clone(model, true), act.bale, state)

        case Act.LIBRARY_MENU:
            return Buzz.libraryMenu(clone(model, true), act.bale, state)

        case Act.ROUTE_MENU:
            return Buzz.routeMenu(clone(model, true), act.bale, state)

        case Act.FLEET_MENU:
            return Buzz.fleetMenu(clone(model, true), act.bale, state)

        case Act.SAGA_TORN_STATE_MENU:
            return Buzz.sagaTornStateMenu(clone(model, true), act.bale, state)

        default:
            return model
    }
}

                src: `>> [DESYNC] Flagging failed targets as DESYNCHRONIZED. Staging buffers unlinked.`,
            });
            await ste.hunt(ActMnu.FLEET_MENU, {});
            return cpy;
        }

        case '3. COMPENSATE & ROLLBACK COMPLETED (Restore from backups)': {
            await ste.hunt(ActCns.UPDATE_CONSOLE, {
                idx: 'cns00',
                src: '>> [COMPENSATE] Rolling back completed targets in reverse order...',
            });
            const fs = require('fs-extra');
            for (const target of [...(sagaJournal as any).targetsCompleted].reverse()) {
                const bkp = (sagaJournal as any).backups?.[target];
                if (bkp && (await fs.pathExists(bkp))) {
                    await fs.remove(target).catch(() => {});
                    await fs.rename(bkp, target).catch(() => {});
                    await ste.hunt(ActCns.UPDATE_CONSOLE, { idx: 'cns00', src: `>> [RESTORED] Reverted: ${target}` });
                }
            }
            await ste.hunt(ActMnu.FLEET_MENU, {});
            return cpy;
        }

        case '4. ABORT SAGA & PRESERVE PARTIAL STATE': {
            await ste.hunt(ActCns.UPDATE_CONSOLE, {
                idx: 'cns00',
                src: '>> [ABORT] Preserving partial state. Active journal sealed as TORN.',
            });
            await ste.hunt(ActMnu.FLEET_MENU, {});
            return cpy;
        }

        case '5. DUMP FORENSIC SAGA DOSSIER TO cns00': {
            await ste.hunt(ActCns.UPDATE_CONSOLE, { idx: 'cns00', src: '>> [DOSSIER DUMP] ================================' });
            const lines = JSON.stringify(sagaJournal, null, 2).split('\n');
            for (const line of lines) {
                await ste.hunt(ActCns.UPDATE_CONSOLE, { idx: 'cns00', src: line });
            }
            await ste.hunt(ActCns.UPDATE_CONSOLE, { idx: 'cns00', src: '>> ================================================' });
            break;
        }

        case '<-- RETURN TO MAIN FLEET MENU':
            await ste.hunt(ActMnu.FLEET_MENU, {});
            return cpy;
    }

    setTimeout(async () => {
        await ste.hunt(ActMnu.SAGA_TORN_STATE_MENU, { dat: sagaJournal });
    }, 333);

    return cpy;
};
