### II. ARCHITECTURAL MISSION & REQUIREMENTS

When an operator inputs a directive focus label in the Blessed TUI containing spaces or uppercase characters (e.g., `moderize media pipeline`), the generated `.cpn` divider filename and the resolver response currently preserve raw whitespace (e.g., `001.moderize media pipeline.cpn`).

This task implements **Automatic Directive Divider Slugification (`DIRECTIVE_DIVIDER_SLUGIFICATION`)**:

1. **Slugification Normalizer (`directive.buzz.ts`):**
* Transform the incoming `bal.idx` / `label` using `slugify(rawLabel, { lower: true, strict: true, trim: true })` (leveraging the already installed `slugify` package in `apps/995.library/package.json`) with a resilient regex fallback (`replace(/[\s_]+/g, '-').replace(/[^a-z0-9-]/g, '').replace(/-+/g, '-').replace(/^-+|-+$/g, '')`).


* A label entered as `moderize media pipeline` must produce `moderize-media-pipeline`.




2. **Divider Filename Construction:**
* Construct the divider file strictly as `${pad3(currentSeq)}.${slugLabel}.cpn` (e.g., `001.moderize-media-pipeline.cpn`).




3. **Resolver Payload Alignment:**
* In `bal.slv({ drtBit: { ... } })`, ensure `dat.label` reflects the slugified label and `dat.files[0]` reflects `${pad3(currentSeq)}.${slugLabel}.cpn`.




4. **TUI Menu Ingress Pre-Normalization (`menu.library.ts`):**
* Normalize the input string in `menu.library.ts` before dispatching `ActDrt.CREATE_DIRECTIVE`.




5. **Automated Test Battery:**
* Add test coverage in `apps/995.library/test/directive.create.test.ts` asserting that labels with spaces, uppercase letters, and non-ASCII/special characters are cleanly slugified.

/**
 * Normalizes a raw focus label string into a clean URL/filename-safe slug.
 * Transforms whitespace, underscores, and special characters into hyphens.
 */
export const toSlug = (raw: string): string => {
    if (!raw) return 'focus'
    try {
        const slug = slugify(raw, { lower: true, strict: true, trim: true })
        if (slug && slug.length > 0) return slug
    } catch {}
    return (
        raw
            .toLowerCase()
            .trim()
            .replace(/[\s_]+/g, '-')
            .replace(/[^a-z0-9-]/g, '')
            .replace(/-+/g, '-')
            .replace(/^-+|-+$/g, '') || 'focus'
    )
}

const isRepoRoot = (dir: string): boolean => {

    }

const findRepoRoot = (startDir: string = process.cwd()): string => {
    const path = require('path')
    let curr = path.resolve(startDir)
    while (curr && curr !== path.dirname(curr)) {
        if (isRepoRoot(curr)) return curr
        curr = path.dirname(curr)
    }
    return process.cwd()
}

export const dayDirective = async (
    cpy: DirectiveModel,
    bal: DirectiveBit,
    ste: State,
): Promise<DirectiveModel> => {
    const fs = require('fs-extra')
    const path = require('path')

    const repoRoot = findRepoRoot()
    const directiveRootDir = path.join(repoRoot, 'data', 'directive')

    try {
        await fs.ensureDir(directiveRootDir)

        const entries = await fs.readdir(directiveRootDir, { withFileTypes: true })
        let maxIndex = -1

        for (const entry of entries) {
            if (!entry.isDirectory()) continue
            const match = entry.name.match(/^day-(\d+)$/i)
            if (match) {
                const num = parseInt(match[1], 10)
                if (num > maxIndex) {
                    maxIndex = num
                }
            }
        }

        const nextIndex = maxIndex + 1
        const daySlug = `day-${String(nextIndex).padStart(3, '0')}`
        const targetDayDir = path.join(directiveRootDir, daySlug)

        if (ste) {
            await ste.hunt(ActCns.UPDATE_CONSOLE, {
                idx: 'cns00',
                src: `>> [DIRECTIVE] Provisioning shift ${daySlug}...`,
            })
        }

        await fs.ensureDir(targetDayDir)

        const templateFiles = [
            '000.focus.cpn',
            '000.xxxx.jules',
            '000.xxxxx.ag-test',
            '000.xxxxx.gravity',
        ]

        for (const filename of templateFiles) {
            const destPath = path.join(targetDayDir, filename)
            await fs.writeFile(destPath, '', 'utf8')
            if (ste) {
                await ste.hunt(ActCns.UPDATE_CONSOLE, {
                    idx: 'cns00',
                    src: `>> [OK] writing data/directive/${daySlug}/${filename}`,
                })
            }
        }

        const relOutput = path.relative(repoRoot, targetDayDir).replace(/\\/g, '/')

        if (ste) {
            await ste.hunt(ActCns.UPDATE_CONSOLE, {
                idx: 'cns00',
                src: `>> [OK] Successfully created directive shift: ${relOutput}`,
            })
        }

        if (bal && bal.slv != null) {
            bal.slv({
                drtBit: {
                    idx: 'day-directive',
                    src: relOutput,
                    val: nextIndex,
                    dat: {
                        dayFolder: daySlug,
                        files: templateFiles,
                    },
                },
                dayBit: {
                    idx: daySlug,
                },
            })
        }
    } catch (err: any) {
        const errorMsg = err instanceof Error ? err.message : String(err)
        if (ste) {
            await ste.hunt(ActCns.UPDATE_CONSOLE, {
                idx: 'cns00',
                src: `:: [FAIL] Error in dayDirective: ${errorMsg}`,
            })
        }
        if (bal && bal.slv != null) {
            bal.slv({
                drtBit: {
                    idx: 'day-directive-error',
                    src: errorMsg,
                },
            })
        }
    }

    return cpy
}

export const createDirective = async (
    cpy: DirectiveModel,
    bal: DirectiveBit,
    ste: State,
): Promise<DirectiveModel> => {
    const fs = require('fs-extra')
    const path = require('path')

    const repoRoot = findRepoRoot()
    const targetDay = (bal?.src || 'day-000').replace(/[[\]]/g, '').trim()
    const rawLabel = (bal?.idx || 'focus').trim()
    const slugLabel = toSlug(rawLabel)
    const phaseCount = typeof bal?.val === 'number' && bal.val > 0 ? bal.val : 1

    const targetDayDir = path.join(repoRoot, 'data', 'directive', targetDay)
    const templateDayDir = path.join(repoRoot, 'data', 'directive', 'day-000')

    try {
        await fs.ensureDir(targetDayDir)

        if (ste) {
            await ste.hunt(ActCns.UPDATE_CONSOLE, {
                idx: 'cns00',
                src: `>> [DIRECTIVE] Scaffolding stack in ${targetDay} (${phaseCount} phases)...`,
            })
        }

        // 1. Calculate next monotonic prefix
        const existingEntries = await fs.readdir(targetDayDir)
        let maxPrefix = -1

        for (const entry of existingEntries) {
            const match = entry.match(/^(\d{3})\./)
            if (match) {
                const num = parseInt(match[1], 10)
                if (num > maxPrefix) {
                    maxPrefix = num
                }
            }
        }

        let currentSeq = maxPrefix + 1
        const createdFiles: string[] = []

        const readTemplateContent = async (baseNames: string[]): Promise<string> => {
            const searchDirs = [templateDayDir, targetDayDir]
            for (const dir of searchDirs) {
                for (const baseName of baseNames) {
                    const candidate = path.join(dir, baseName)
                    if (await fs.pathExists(candidate)) {
                        return await fs.readFile(candidate, 'utf8')
                    }
                }
            }
            return ''
        }

        // 2. Provision slugified focus divider (.cpn)
        const focusTemplateContent = await readTemplateContent([
            '000.focus.cpn',
            'focus.cpn',
        ])

        const dividerFileName = `${pad3(currentSeq)}.${slugLabel}.cpn`
        const dividerFilePath = path.join(targetDayDir, dividerFileName)
        await fs.writeFile(dividerFilePath, focusTemplateContent, 'utf8')
        createdFiles.push(dividerFileName)

        if (ste) {
            await ste.hunt(ActCns.UPDATE_CONSOLE, {
                idx: 'cns00',
                src: `>> [OK] writing data/directive/${targetDay}/${dividerFileName}`,
            })
        }

        // 3. Read Jules and Gravity/AG-Test templates
        const julesTemplateContent = await readTemplateContent([
            '000.xxxx.jules.md',
            '000.xxxx.jules',
            '000.jules.md',
        ])

        const gravityTemplateContent = await readTemplateContent([
            '000.xxxxx.gravity.md',
            '000.xxxxx.gravity',
            '000.xxxxx.ag-test.md',
            '000.xxxxx.ag-test',
        ])

        let checkExt = 'gravity.md'
        if (
            (await fs.pathExists(path.join(templateDayDir, '000.xxxxx.ag-test.md'))) &&
            !(await fs.pathExists(path.join(templateDayDir, '000.xxxxx.gravity.md')))
        ) {
            checkExt = 'ag-test.md'
        }

        // 4. Provision sequential phase pairs
        for (let p = 1; p <= phaseCount; p++) {
            const phaseWord = getPhaseWord(p)

            currentSeq++
            const julesFileName = `${pad3(currentSeq)}.phase-${phaseWord}.jules.md`
            const julesFilePath = path.join(targetDayDir, julesFileName)
            await fs.writeFile(julesFilePath, julesTemplateContent, 'utf8')
            createdFiles.push(julesFileName)

            if (ste) {
                await ste.hunt(ActCns.UPDATE_CONSOLE, {
                    idx: 'cns00',
                    src: `>> [OK] writing data/directive/${targetDay}/${julesFileName}`,
                })
            }

            currentSeq++
            const gravityFileName = `${pad3(currentSeq)}.phase-${phaseWord}.${checkExt}`
            const gravityFilePath = path.join(targetDayDir, gravityFileName)
            await fs.writeFile(gravityFilePath, gravityTemplateContent, 'utf8')
            createdFiles.push(gravityFileName)

            if (ste) {
                await ste.hunt(ActCns.UPDATE_CONSOLE, {
                    idx: 'cns00',
                    src: `>> [OK] writing data/directive/${targetDay}/${gravityFileName}`,
                })
            }
        }

        const relOutput = path.relative(repoRoot, targetDayDir).replace(/\\/g, '/')

        if (ste) {
            await ste.hunt(ActCns.UPDATE_CONSOLE, {
                idx: 'cns00',
                src: `>> [OK] Directive stack created in ${relOutput} (${createdFiles.length} files).`,
            })
        }

        if (bal && bal.slv != null) {
            bal.slv({
                drtBit: {
                    idx: 'create-directive',
                    src: relOutput,
                    val: createdFiles.length,
                    dat: {
                        targetDay,
                        label: slugLabel,
                        files: createdFiles,
                    },
                },
            })
        }
    } catch (err: any) {
        const errorMsg = err instanceof Error ? err.message : String(err)
        if (ste) {
            await ste.hunt(ActCns.UPDATE_CONSOLE, {
                idx: 'cns00',
                src: `:: [FAIL] Error in createDirective: ${errorMsg}`,
            })
        }
        if (bal && bal.slv != null) {
            bal.slv({
                drtBit: {
                    idx: 'create-directive-error',
                    src: errorMsg,
                },
            })
        }
    }

    return cpy
}
