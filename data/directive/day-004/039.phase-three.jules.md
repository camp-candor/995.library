import type Console from './fce/console.interface'
import type { TelemetryFrame, RingBufferQuery, RingBufferStats } from './fce/ring-buffer.interface'

export class ConsoleRingBuffer {
    public readonly capacity: number
    private buffer: Array<TelemetryFrame | null>
    private head: number = 0
    private count: number = 0
    private sequence: number = 0

    constructor(capacity: number = 10000) {
        this.capacity = Math.max(1, capacity)
        this.buffer = new Array(this.capacity).fill(null)
    }

    public push(rawText: string, metadata?: Partial<TelemetryFrame>): TelemetryFrame {
        this.sequence++
        const clean = rawText.replace(/[^\x20-\x7E]/g, '').trim()
        const frame: TelemetryFrame = {
            id: this.sequence,
            timestamp: new Date().toISOString(),
            raw: rawText,
            clean,
            repo: metadata?.repo,
            stage: metadata?.stage,
            status: metadata?.status,
        }

        this.buffer[this.head] = frame
        this.head = (this.head + 1) % this.capacity
        if (this.count < this.capacity) {
            this.count++
        }

        return frame
    }

    public toArray(): TelemetryFrame[] {
        if (this.count === 0) return []
        if (this.count < this.capacity) {
            return this.buffer.slice(0, this.count) as TelemetryFrame[]
        }
        const tail = this.head
        const part1 = this.buffer.slice(tail) as TelemetryFrame[]
        const part2 = this.buffer.slice(0, tail) as TelemetryFrame[]
        return part1.concat(part2)
    }

    public filter(query: RingBufferQuery): TelemetryFrame[] {
        const frames = this.toArray()
        const limit = query.limit && query.limit > 0 ? query.limit : frames.length

        const results: TelemetryFrame[] = []
        for (let i = frames.length - 1; i >= 0 && results.length < limit; i--) {
            const frame = frames[i]
            if (query.repo && frame.repo !== query.repo) continue
            if (query.stage && frame.stage !== query.stage) continue
            if (query.status && frame.status !== query.status) continue
            if (query.query && !frame.clean.toLowerCase().includes(query.query.toLowerCase())) continue
            results.push(frame)
        }

        return results.reverse()
    }

    public getStats(): RingBufferStats {
        return {
            capacity: this.capacity,
            size: this.count,
            head: this.head,
            totalIngested: this.sequence,
        }
    }

    public clear(): void {
        this.buffer.fill(null)
        this.head = 0
        this.count = 0
    }
}

export class ConsoleModel implements Console {
    ring = new ConsoleRingBuffer(10000)
    activeFilter: RingBufferQuery = {}
}

function wrapConsoleLine(text: string, maxWidth: number): string[] {
    if (!text) return ['']

    if (/^(?:>>\s*)?[-=_*~]{10,}$/.test(text)) {
        return [text.slice(0, maxWidth)]
    }

    if (text.length <= maxWidth) {
        return [text]
    }

    const prefixMatch = text.match(/^(\s*(?:>>|\*|-|•)\s*)/)
    const basePrefix = prefixMatch ? prefixMatch[1] : ''
    const contPrefix = basePrefix.trim() === '>>' ? '>>   ' : '   '

    const lines: string[] = []
    let remaining = text
    let isFirst = true

    while (remaining.length > 0) {
        const currentPrefix = isFirst ? '' : contPrefix
        const effectiveMax = Math.max(10, maxWidth - currentPrefix.length)

        if (remaining.length <= effectiveMax) {
            lines.push(currentPrefix + remaining)
            break
        }

        let breakIdx = remaining.lastIndexOf(' ', effectiveMax)
        if (breakIdx <= 0 || breakIdx < effectiveMax * 0.4) {
            breakIdx = effectiveMax
        }

        const chunk = remaining.slice(0, breakIdx).trimEnd()
        lines.push(currentPrefix + chunk)
        remaining = remaining.slice(breakIdx).trimStart()
        isFirst = false
    }

    return lines
}

function wrapConsoleContent(content: string, maxWidth: number): string[] {
    if (!content) return ['']
    const paragraphs = content.split(/\r?\n/)
    const result: string[] = []
    for (const p of paragraphs) {
        result.push(...wrapConsoleLine(p, maxWidth))
    }
    return result
}

            terminal.screen.render()
        }
    })
}

    bal: ConsoleBit,

    const dat: TermBit | undefined = bit?.cnsBit?.dat

        let maxWidth = 48
        if (typeof dat.bit.width === 'number' && dat.bit.width > 4) {
            maxWidth = dat.bit.width - 3
        } else if (
            dat.bit.lpos &&
            typeof dat.bit.lpos.xl === 'number' &&
            typeof dat.bit.lpos.xi === 'number'
        ) {
            const calculated = dat.bit.lpos.xl - dat.bit.lpos.xi - 3
            if (calculated > 10) maxWidth = calculated
        } else if (terminal?.screen?.cols && terminal.screen.cols > 10) {

                20,
                Math.floor((terminal.screen.cols * 8) / 12) - 3,
            )

        for (const line of lines) {
            dat.bit.log(line)
        }

        scheduleScreenRender(terminal)
    }

    if (bal?.slv != null) bal.slv({ cnsBit: { idx: 'update-console' } })
    return cpy
}
