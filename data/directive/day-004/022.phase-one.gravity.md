  leave: sinon.spy(() => callOrder.push("leave")),

  slv: sinon.spy()

  console.error("FAIL: screen.leave() must be invoked before editor spawns. Got: " + callOrder);

  console.error("FAIL: screen.enter() and render() must be invoked after editor exits. Got: " + callOrder);
  process.exit(1);

  console.error("FAIL: bal.slv was not resolved with exit code 0.");
  process.exit(1);
