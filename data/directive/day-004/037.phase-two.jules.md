### I. Architectural Context & Invariants

1. **The TTY Raw-Mode Trap & Banned CLI Prompts:**
* In `apps/995.library`, the process takes exclusive ownership of the TTY interface via `process.stdin.setRawMode(true)` and renders inside an alternate screen buffer (`\x1b[?1049h`).


* Spawning detached CLI prompt packages (such as `inquirer`, `@clack/prompts`, `enquirer`, or Node's `readline`) inside an active Blessed process creates fatal stream contention, unbinds window listeners, corrupts curses coordinates, and crashes the terminal.


* All interactive user input must be captured natively through curses widgets: single-line text input through `84.input.unit` (`blessed.textbox` / `OPEN_INPUT`) and multi-option vertical selection through `85.choice.unit` (`blessed.form` + `blessed.button` / `OPEN_CHOICE`).




2. **Atomic Lifecycle & Zero Listener Leakage:**
* Widgets mounted in the Control Axis (Columns 0–4) must bind navigation listeners locally rather than leaking global screen callbacks.


* `OPEN_CHOICE` must map vertical navigation (`['up', 'k']` and `['down', 'j']`) and `Escape` to cancel. Upon submission or cancellation, it must deregister all keys via `screen.unkey`, invoke `form.destroy()`, and trigger `screen.render()` before resolving `bal.slv`.


* `OPEN_INPUT` must bind `Escape` for cancellation. Upon submission or cancellation, it must unbind listeners, destroy both the title line and textbox widgets, re-render the screen, and resolve `bal.slv` cleanly with `{ putBit: { idx: 'open-input' } }` or `{ putBit: { idx: 'open-input-cancel' } }`.




3. **Repository Laws (`AGENTS.md`):**
* Immutability on `apps/995.library/995.library/**` is **permanently lifted** in this repository.

import type { InputModel } from '../input.model'
import type InputBit from '../fce/input.bit'
import type State from '../../99.core/state'
import type NetBit from '../../81.grid.unit/fce/net.bit'
import * as Color from '../../val/console-color'

export const initInput = (cpy: InputModel, bal: InputBit, _ste: State) => {
    if (bal?.slv != null) bal.slv({ putBit: { idx: 'init-input' } })
    return cpy
}

export const updateInput = (cpy: InputModel, bal: InputBit, _ste: State) => {
    if (bal?.slv != null) bal.slv({ putBit: { idx: 'update-input' } })
    return cpy
}

export const openInput = (cpy: InputModel, bal: InputBit, ste: State) => {
    const terminal = ste?.value?.terminal
    const blessed = terminal?.blessed
    const screen = terminal?.screen

    // Headless fallback for automated test gauntlets and CI runners
    if (!screen || !blessed) {
        if (bal?.slv != null) {
            bal.slv({
                putBit: {
                    idx: 'open-input-headless',
                    src: bal?.src || '',
                    val: 0,
                },
            })
        }
        return cpy
    }

    const dat = {
        idx: 'input-bit',
        clr0: Color.GREEN,
        clr1: Color.CYAN,
        ...(bal?.dat || {}),
    }

    const net: NetBit = bal?.net || {
        left: 0,
        top: 0,
        width: '100%',
        height: 3,
    }

    const labelText = bal?.txt || 'input below'

    const title = blessed.textbox({
        parent: screen,
        name: 'input-title',
        keys: false,
        left: net.left,
        top: typeof net.top === 'number' ? net.top : 1,
        width: net.width,
        height: 2,
        content: labelText,
        border: { type: 'line' },
        style: {
            fg: dat.clr0,
            border: { fg: dat.clr0 },
        },
    })

    const input = blessed.textbox({
        parent: screen,
        name: 'input-body',
        input: true,
        keys: true,
        mouse: true,
        left: net.left,
        top: typeof net.top === 'number' ? net.top + 2 : 3,
        width: net.width,
        height: typeof net.height === 'number' ? Math.max(1, net.height - 2) : 1,
        inputOnFocus: true,
        style: {
            bg: dat.clr1,
            fg: Color.BLACK,
            focus: { bg: Color.YELLOW, fg: Color.BLACK },
            hover: { bg: Color.YELLOW, fg: Color.BLACK },
        },
    })

    let cleanedUp = false
    const cleanup = () => {
        if (cleanedUp) return
        cleanedUp = true

        try {
            input.unkey(['escape'], handleCancel)
            title.destroy()
            input.destroy()
            screen.render()
        } catch {
            // Teardown defensive catch
        }
    }

    const handleCancel = () => {
        cleanup()
        if (bal?.slv != null) {
            bal.slv({
                putBit: {
                    idx: 'open-input-cancel',
                    src: '',
                    val: -1,
                },
            })
        }
    }

    input.key(['escape'], handleCancel)

    input.on('submit', (src: string) => {
        const valStr = src || ''
        cleanup()
        if (bal?.slv != null) {
            bal.slv({
                putBit: {
                    idx: 'open-input',
                    src: valStr,
                    val: 0,
                },
            })
        }
    })

    input.focus()
    screen.render()

    return cpy
}

import type { ChoiceModel } from '../choice.model'
import type ChoiceBit from '../fce/choice.bit'
import type State from '../../99.core/state'

export const initChoice = (cpy: ChoiceModel, bal: ChoiceBit, _ste: State) => {

    }
    return cpy
}

export const updateChoice = (cpy: ChoiceModel, bal: ChoiceBit, _ste: State) => {
    if (bal?.slv != null) {
        bal.slv({ chcBit: { idx: 'update-choice' } })
    }
    return cpy
}

export const openChoice = (cpy: ChoiceModel, bal: ChoiceBit, ste: State) => {
    const terminal = ste?.value?.terminal
    const blessed = terminal?.blessed
    const screen = terminal?.screen

    // Defensive headless fallback for automated runners and CI environments
    if (!screen || !blessed) {
        if (bal?.slv != null) {
            bal.slv({
                chcBit: {
                    idx: 'open-choice-headless',
                    src: bal?.lst && bal.lst.length > 0 ? bal.lst[0] : '',
                    val: 0,
                },
            })
        }
        return cpy
    }

    const dat: any = { idx: 'choice-bit', clr0: Color.BLACK, clr1: Color.YELLOW }
    for (const key in bal?.dat) {
        dat[key] = bal.dat[key]
    }

    const net: NetBit = bal?.net || {
        left: 0,
        top: 0,
        width: '100%',
        height: '100%',
    }

    const form = blessed.form({
        parent: screen,
        keys: false,
        mouse: true,
        left: net.left,
        top: net.top,
        width: net.width,
        height: net.height,
        bg: dat.clr0,
        content: '',
    })

    const buttons: any[] = []
    const itemList: string[] = bal?.lst || []

    itemList.forEach((itemText: string, index: number) => {
        const btn = blessed.button({
            parent: form,
            mouse: true,
            keys: true,
            shrink: true,
            padding: {
                left: 2,
                right: 1,
            },
            left: 0,
            top: index,
            height: 1,
            width: '100%',
            name: itemText,
            content: itemText,
            style: {
                bg: dat.clr1,
                focus: {
                    bg: 'red',
                },
                hover: {
                    bg: 'red',
                },
            },
        })

        btn.on('press', () => {
            form.submit()
        })

        btn.on('focus', () => {
            form._selected = btn
            if (dat.cb) {
                dat.cb(itemText)
            }
        })

        buttons.push(btn)
    })

    // Vertical keyboard navigation mappings (Up/Down + Vim k/j)
    const handleUp = () => form.focusPrevious()
    const handleDown = () => form.focusNext()
    const handleCancel = () => form.cancel()

    screen.key(['up', 'k'], handleUp)
    screen.key(['down', 'j'], handleDown)
    screen.key(['escape'], handleCancel)

    let cleanedUp = false
    const cleanup = () => {
        if (cleanedUp) return
        cleanedUp = true

        // 1. Explicitly unbind all registered directional and cancel navigation keys
        screen.unkey(['up', 'k'], handleUp)
        screen.unkey(['down', 'j'], handleDown)
        screen.unkey(['escape'], handleCancel)

        // 2. Detach and destroy widget subtree to liberate terminal memory
        form.destroy()

        // 3. Request coordinate repaint
        screen.render()
    }

    form.on('submit', () => {
        let selected = form._selected
        if (!selected && buttons.length > 0) {
            selected = buttons[0]
        }

        const src = selected ? selected.content : ''
        const val = selected ? selected.index - 1 : 0

        cleanup()

        if (bal?.slv != null) {
            bal.slv({ chcBit: { idx: 'open-choice', src, val } })
        }
    })

    form.on('cancel', () => {
        cleanup()

        if (bal?.slv != null) {
            bal.slv({ chcBit: { idx: 'open-choice-cancel', src: '', val: -1 } })
        }
    })

    if (buttons.length > 0 && typeof buttons[0].focus === 'function') {
        buttons[0].focus()
    }

    screen.render()
    return cpy
}

export const keyChoice = (cpy: ChoiceModel, bal: ChoiceBit, ste: State) => {
    const terminal = ste?.value?.terminal
    const blessed = terminal?.blessed
    const screen = terminal?.screen

    if (!screen || !blessed) {
        if (bal?.slv != null) {
            bal.slv({ chcBit: { idx: 'key-choice-headless' } })
        }
        return cpy
    }

    const net: NetBit = bal?.net || { left: 0, top: 0, width: '100%', height: '100%' }

    blessed.listbar({
        parent: screen,
        keys: true,
        left: net.left,
        top: net.top,
        width: net.width,
        height: net.height,
        style: { item: { fg: 'yellow' }, selected: { fg: 'yellow' } },
        commands: {
            Exit: {
                keys: ['C-c', 'escape'],
                callback: () => process.exit(0),
            },
        },
    })

    screen.render()
    if (bal?.slv != null) {
        bal.slv({ chcBit: { idx: 'key-choice' } })
    }
    return cpy
}
