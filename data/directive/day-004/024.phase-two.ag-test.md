#### Level 4: Canonical Non-Mutation & Scratchpad Lifecycle Controls

node -e '
const fs = require("fs-extra");
const path = require("path");
const { LibraryModel } = require("./apps/995.library/dist/995.library/00.library.unit/library.model");
const {
  stageLockfileConflict,
  purgeLockfileScratchpad
} = require("./apps/995.library/dist/995.library/00.library.unit/buz/lockfile.buzz");

async function run() {
  const sandbox = path.resolve(process.cwd(), "scratch_antigravity_lockfile");
  const canonicalFile = path.join(sandbox, "versions.json");
  await fs.ensureDir(sandbox);

  const initialCanonical = {
    schema_version: 1,
    library_version: "1.0.12",
    repositories: {
      "repo-a": { pinned_sha: "1111111111111111111111111111111111111111" }
    }
  };
  await fs.writeJson(canonicalFile, initialCanonical);

  const model = new LibraryModel();
  let resolvedBit = null;

  const bal = {
    src: canonicalFile,
    dat: {
      localManifest: initialCanonical,
      remoteManifest: {
        schema_version: 1,
        library_version: "1.0.13",
        repositories: {
          "repo-a": { pinned_sha: "2222222222222222222222222222222222222222" }
        }
      }
    },
    slv: (res) => { resolvedBit = res; }
  };

  await stageLockfileConflict(model, bal, null);

  if (!resolvedBit || resolvedBit.libBit.idx !== "stage-lockfile-conflict") {
    console.error("FAIL: stageLockfileConflict did not resolve with stage-lockfile-conflict");
    process.exit(1);
  }

  const scratchPath = resolvedBit.libBit.src;
  if (!scratchPath.includes("data/lockfile") && !scratchPath.includes("data\\lockfile")) {
    console.error("FAIL: Scratchpad file must be staged under data/lockfile/. Got: " + scratchPath);
    process.exit(1);
  }
  if (!await fs.pathExists(scratchPath)) {
    console.error("FAIL: Staged scratchpad file does not exist on disk.");
    process.exit(1);
  }

  // Canonical file must be bit-identical to initial state
  const canonicalAfter = await fs.readJson(canonicalFile);
  if (JSON.stringify(canonicalAfter) !== JSON.stringify(initialCanonical)) {
    console.error("FAIL: Canonical versions.json was modified during conflict staging!");
    process.exit(1);
  }

  // Verify purge cleans up temporary scratchpad
  let purgeResolved = null;
  await purgeLockfileScratchpad(model, { src: scratchPath, slv: (r) => { purgeResolved = r; } }, null);

  if (await fs.pathExists(scratchPath)) {
    console.error("FAIL: Scratchpad was not unlinked by purgeLockfileScratchpad.");
    process.exit(1);
  }

  await fs.remove(sandbox);
  console.log(">> [PASS] Level 4: Canonical immutability and scratchpad purge lifecycle certified.");
}

run().catch((err) => {
  console.error("FAIL Level 4:", err);
  process.exit(1);
});
